CREATE TABLE IF NOT EXISTS IPL_DWH.RAW.RAW_IPL_JSON (

    PAYLOAD     VARIANT
)
COMMENT = 'One row per Cricsheet match JSON; PAYLOAD holds the full document';

-- ---------------------------------------------------------------------------
-- 8. Unzip stored procedure  (ZIP on stage → individual JSONs on stage)
-- ---------------------------------------------------------------------------
CREATE OR REPLACE PROCEDURE IPL_DWH.RAW.SP_UNZIP_CRICSHEET("ZIP_FILE_NAME" VARCHAR)
RETURNS VARCHAR
LANGUAGE PYTHON
RUNTIME_VERSION = '3.10'
PACKAGES = ('snowflake-snowpark-python')
HANDLER = 'main'
EXECUTE AS OWNER
AS
$$
import io
import os
import zipfile

SRC_STAGE  = "@IPL_DWH.RAW.STG_IPL_ZIP"
DEST_STAGE = "@IPL_DWH.RAW.STG_IPL_JSON"

def main(session, zip_file_name: str) -> str:
    src_path = f"{SRC_STAGE}/{zip_file_name}"

    with session.file.get_stream(src_path) as fh:
        zip_bytes = fh.read()
    buf = io.BytesIO(zip_bytes)

    uploaded = skipped = failed = 0
    errors = []

    with zipfile.ZipFile(buf) as zf:
        for member in zf.infolist():
            name = member.filename
            if (member.is_dir()
                or not name.lower().endswith(".json")
                or name.startswith("__MACOSX/")
                or os.path.basename(name).startswith(".")):
                skipped += 1
                continue
            try:
                with zf.open(member, "r") as src:
                    data = src.read()
                flat_name = os.path.basename(name)
                file_obj  = io.BytesIO(data)
                file_obj.seek(0)
                session.file.put_stream(
                    file_obj,
                    f"{DEST_STAGE}/{flat_name}",
                    auto_compress = True,
                    overwrite     = True,
                )
                uploaded += 1
            except Exception as e:
                failed += 1
                if len(errors) < 5:
                    errors.append(f"{name}: {type(e).__name__}: {e}")

    msg = (f"Source: {src_path} | Uploaded: {uploaded} | "
           f"Skipped: {skipped} | Failed: {failed} -> {DEST_STAGE}")
    if errors:
        msg += " | Sample errors: " + " || ".join(errors)
    return msg
$$;
